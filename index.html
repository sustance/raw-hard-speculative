const USERNAME = 'sustance'; const REPO = 'raw-hard-speculative'; const FOLDER = 'seeds'; const LINES_TO_EXTRACT = 4; async function fetchSeeds() { const container = document.getElementById('seed-list'); try { // Targets your exact repository path const url = `https://github.com{USERNAME}/${REPO}/contents/${FOLDER}`; const response = await fetch(url); if (!response.ok) throw new Error(`GitHub API Error: ${response.status}`); const items = await response.json(); container.innerHTML = ''; for (const item of items) { if (item.type === 'file') { const fileContentResponse = await fetch(item.download_url); const fullText = await fileContentResponse.text(); // Extracts the top lines as a summary description const lines = fullText.split(/\r?\n/).slice(0, LINES_TO_EXTRACT); const description = lines.join('\n'); const card = document.createElement('div'); card.className = 'file-card'; card.innerHTML = ` ${item.name}
${description || '(No content inside file)'}
`; container.appendChild(card); } } } catch (error) { console.error(error); container.innerHTML = `
Error: ${error.message}
`; } } fetchSeeds();
